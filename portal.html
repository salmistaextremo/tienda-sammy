async function loadMyOrders(userId, userEmail) { const list = document.getElementById('ordersList'); try { // Consulta directa usando el ID y el correo del usuario logueado const { data: orders, error } = await supabaseClient .from('orders') .select('*, installments(amount_bs, amount_usdt)') .or(`user_id.eq.${userId},customer_email.eq.${userEmail}`) .order('created_at', { ascending: false }); if (error) throw error; if (!orders || orders.length === 0) { list.innerHTML = `

Aún no tienes pedidos

Cuando realices una compra, aparecerá aquí con todos los detalles.

Ir a la tienda
`; return; } list.innerHTML = orders.map(o => { const shortId = String(o.id).slice(0,8).toUpperCase(); const date = new Date(o.created_at).toLocaleDateString('es-VE', { day: '2-digit', month: 'long', year: 'numeric' }); let abonadoBs = 0, abonadoUsdt = 0; if (o.installments && o.installments.length > 0) { o.installments.forEach(i => { abonadoBs += Number(i.amount_bs || 0); abonadoUsdt += Number(i.amount_usdt || 0); }); } const totalBs = Number(o.total_bs); const saldoBs = Math.max(0, totalBs - abonadoBs); const isPaid = saldoBs <= 0.1; const statusClass = `status-${o.status}`; return `
Pedido #${shortId}
${date}
${o.status.replace('_', ' ')}
Modalidad ${o.payment_plan}
Total del Pedido ${totalBs.toLocaleString('es-VE')} Bs ≈ ${Number(o.total_usdt).toFixed(2)} USDT
Total Abonado ${abonadoBs.toLocaleString('es-VE')} Bs ≈ ${abonadoUsdt.toFixed(2)} USDT
Saldo Pendiente ${isPaid ? `Pagado totalmente` : `${saldoBs.toLocaleString('es-VE')} Bs ≈ ${(Number(o.total_usdt) - abonadoUsdt).toFixed(2)} USDT` }
`; }).join(''); } catch (err) { console.error(err); list.innerHTML = `

Error al cargar pedidos: ${err.message}

`; } }
